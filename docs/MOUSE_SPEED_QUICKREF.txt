│ 脚本模板                                                          │
└─────────────────────────────────────────────────────────────────┘

  @Override
  public boolean run(YourConfig config) {
      // ① 启动时设置速度
      Microbot.naturalMouse.setCustomMouseSpeedMs(100);
      
      mainLoop = true;
      while (mainLoop) {
          // ② 你的脚本逻辑
          // 所有鼠标移动都会使用自定义速度
      }
      
      // ③ 清理：恢复默认
      Microbot.naturalMouse.setCustomMouseSpeedMs(null);
      return true;
  }
  
  @Override
  public void shutdown() {
      super.shutdown();
      // ④ 确保清理
      Microbot.naturalMouse.setCustomMouseSpeedMs(null);
  }

  @ConfigItem(

      name = "使用自定义速度",
      description = "启用后将覆盖 Rs2Antiban 的自动速度"
  )
  default boolean useCustomSpeed() {
      return false;
  }
  
  @ConfigItem(
      keyName = "mouseSpeedMs",
      name = "鼠标速度 (ms)",
      description = "数值越小越快。推荐: 60-200"
  )
  default int mouseSpeedMs() {
      return 100;
  }

  ✓ 线程安全：可在任意线程调用
  ✓ 即时生效：下次鼠标移动立即使用新速度
  ✓ 完全绕过：不需要启用或配置 Rs2Antiban
  ✓ 必须清理：shutdown() 中恢复为 null
  
  ✗ 避免极值：< 50ms 过快不自然，> 300ms 过慢
  ✗ 避免遗忘：不清理会影响其他脚本
